import { describe, it } from 'vitest';
import { hyperMap } from './hyper';

// Does a JOINTLY-FIT hyperbolic embedding beat the real incumbent?
// Incumbent is NOT bare cosine — it is content + graph expansion, which is what
// the live retriever actually does (semantic embedding + graphExpand). Beating
// bare cosine would be a strawman win.
const D = 128;
function mulberry32(a:number){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}
function gauss(r:()=>number){const u1=Math.max(r(),1e-12),u2=r();return Math.sqrt(-2*Math.log(u1))*Math.cos(2*Math.PI*u2);}
const unit=(v:number[])=>{const n=Math.hypot(...v)||1;return v.map(x=>x/n);};
const dot=(a:number[],b:number[])=>a.reduce((s,x,i)=>s+x*b[i],0);

type Node = { id:string; depth:number; content:number[]; parent:string|null };
function corpus(seed:number,{branch=4,depth=4,drift=0.5}={}){
  const r=mulberry32(seed); const nodes:Node[]=[]; const edges:{src:string;dst:string;kind:string;weight:number}[]=[];
  const mk=(parent:Node|null,d:number,path:string)=>{
    const noise=Array.from({length:D},()=>gauss(r));
    const content = parent ? unit(parent.content.map((x,i)=>x*(1-drift)+noise[i]*drift)) : unit(noise);
    const n:Node={id:path,depth:d,content,parent:parent?parent.id:null};
    nodes.push(n); if(parent) edges.push({src:parent.id,dst:n.id,kind:'derivation',weight:1});
    if(d<depth) for(let b=0;b<branch;b++) mk(n,d+1,`${path}.${b}`);
    return n;
  };
  mk(null,0,'r');
  return {nodes,edges};
}

const gp=(id:string)=>id.split('.').slice(0,-2).join('.');
const relevant=(q:Node,n:Node)=>{
  if(n.id===q.id) return false;
  const g=gp(q.id); if(!g) return false;
  return n.id===g || n.id.startsWith(g+'.');   // the whole grandparent subtree
};

function mAP(nodes:Node[], dist:(a:Node,b:Node)=>number, queries:Node[]){
  let tot=0;
  for(const q of queries){
    const rel=nodes.filter(n=>relevant(q,n)).length; if(!rel) continue;
    const ranked=nodes.filter(n=>n.id!==q.id).map(n=>({n,d:dist(q,n)})).sort((a,b)=>a.d-b.d);
    let hits=0,ap=0;
    ranked.forEach((s,i)=>{ if(relevant(q,s.n)){hits++;ap+=hits/(i+1);} });
    tot+=ap/rel;
  }
  return tot/queries.length;
}

function poincare(u:number[],v:number[]){
  let d2=0,nu=0,nv=0;
  for(let i=0;i<u.length;i++){const t=u[i]-v[i];d2+=t*t;nu+=u[i]*u[i];nv+=v[i]*v[i];}
  const a=1-nu,b=1-nv; if(a<=0||b<=0) return Infinity;
  return Math.acosh(1+2*d2/(a*b));
}

describe('DHNG vs Euclidean', () => {
  it('measures a jointly-fit atlas against content+graph recall', () => {
    const {nodes,edges}=corpus(7,{branch:3,depth:6,drift:0.5});
    const byId=new Map(nodes.map(n=>[n.id,n]));
    const rq=mulberry32(99);
    const queries=nodes.filter(n=>n.depth>=2).filter(()=>rq()<0.3);

    // adjacency for the incumbent's graph-expansion tier
    const adj=new Map<string,Set<string>>();
    for(const n of nodes) adj.set(n.id,new Set());
    for(const e of edges){ adj.get(e.src)!.add(e.dst); adj.get(e.dst)!.add(e.src); }

    // --- ARM A: bare cosine (the strawman) ---
    const A=(a:Node,b:Node)=>1-dot(a.content,b.content);

    // --- ARM B: THE REAL INCUMBENT — cosine + 2-hop spreading activation, which
    // is what graphExpand actually does (2-hop BFS with a decayed boost), plus
    // the 1.3x recurrent-cycle boost the live retriever applies. Anything less
    // is a strawman.
    const hop2=new Map<string,Map<string,number>>();
    for(const n0 of nodes){
      const m=new Map<string,number>();
      for(const h1 of adj.get(n0.id)!){ m.set(h1,0.35);
        for(const h2 of adj.get(h1)!) if(h2!==n0.id && !m.has(h2)) m.set(h2,0.65); }
      hop2.set(n0.id,m);
    }
    const B=(a:Node,b:Node)=>{
      const cos=1-dot(a.content,b.content);
      return cos*(hop2.get(a.id)!.get(b.id) ?? 1.0);
    };

    // ablation edges, hoisted so the dim sweep can use them
    const sh=mulberry32(5); const ids=nodes.map(n=>n.id);
    const shuffledEdges=edges.map(()=>({src:ids[Math.floor(sh()*ids.length)],dst:ids[Math.floor(sh()*ids.length)],kind:'derivation',weight:1}));

    // --- ARM C: hyperMap, jointly fit on the SAME edges, swept over dim ---
    const results:Record<number,{c:number;d:number;spread:number}>={};
    for(const dim of [2,4,8,16,32,128]){
      const at=hyperMap(nodes.map(n=>({id:n.id})), edges as never, {dim,epochs:300,seed:42});
      const atR=hyperMap(nodes.map(n=>({id:n.id})), shuffledEdges as never, {dim,epochs:300,seed:42});
      results[dim]={
        c:mAP(nodes,(a,b)=>poincare(at.points[a.id],at.points[b.id]),queries),
        d:mAP(nodes,(a,b)=>poincare(atR.points[a.id],atR.points[b.id]),queries),
        spread:at.stats.depth.max-at.stats.depth.min,
      };
    }
    const atlas=hyperMap(nodes.map(n=>({id:n.id})), edges as never, {dim:2,epochs:300,seed:42});
    const C=(a:Node,b:Node)=>poincare(atlas.points[a.id],atlas.points[b.id]);

    const shuffled=edges.map(()=>({src:ids[Math.floor(sh()*ids.length)],dst:ids[Math.floor(sh()*ids.length)],kind:'derivation',weight:1}));
    const atlasR=hyperMap(nodes.map(n=>({id:n.id})), shuffledEdges as never, {dim:2,epochs:300,seed:42});
    const Dd=(a:Node,b:Node)=>poincare(atlasR.points[a.id],atlasR.points[b.id]);

    const rA=mAP(nodes,A,queries), rB=mAP(nodes,B,queries), rC=mAP(nodes,C,queries), rD=mAP(nodes,Dd,queries);
    console.log(`\n  nodes=${nodes.length} edges=${edges.length} queries=${queries.length} dim=${atlas.dim}`);
    console.log(`  atlas depth: min=${atlas.stats.depth.min.toFixed(3)} mean=${atlas.stats.depth.mean.toFixed(3)} max=${atlas.stats.depth.max.toFixed(3)}`);
    console.log(`  A  bare cosine (strawman)          ${rA.toFixed(4)}`);
    console.log(`  B  cosine + graph expansion  <-REAL ${rB.toFixed(4)}`);
    console.log(`  C  hyperMap H^128 (jointly fit)    ${rC.toFixed(4)}`);
    console.log(`  D  ABLATION shuffled edges         ${rD.toFixed(4)}`);
    console.log(`\n  dim | hyperMap | ablated | structure | radial spread | vs REAL incumbent`);
    for(const dim of [2,4,8,16,32,128]){
      const r=results[dim];
      console.log(`  ${String(dim).padStart(3)} |  ${r.c.toFixed(4)}  | ${r.d.toFixed(4)}  |  ${(r.c-r.d).toFixed(4)}   |    ${r.spread.toFixed(3)}      | ${((100*(r.c-rB)/rB)>0?'+':'')}${(100*(r.c-rB)/rB).toFixed(1)}%`);
    }
    console.log('');
  }, 300000);
});
